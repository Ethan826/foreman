#!/bin/sh
root=$(cd ../.. && pwd -P)
wt=$(pwd -P)
gitdir=$(git rev-parse --path-format=absolute --git-dir)
common=$(git rev-parse --path-format=absolute --git-common-dir)
objects=$(git rev-parse --path-format=absolute --git-path objects)
try() {
  label=$1; target=$2; probe=$3
  if [ -z "$target" ]; then
    echo "$label: unresolved"
    return
  fi
  if err=$(sh -c "$probe" sh "$target" 2>&1 >/dev/null); then
    echo "$label: LANDED exit=0 target=$target"
  else
    status=$?
    echo "$label: refused exit=$status target=$target stderr=$(printf '%s' "$err" | tr '\n' ' ')"
  fi
}
